=== Deno Module

Runtime bindings for Deno APIs.

==== Types

===== `+request+`

Opaque type representing an HTTP request.

===== `+response+`

Opaque type representing an HTTP response.

===== `+addr+`

[source,rescript]
----
type addr = {
  hostname: string,
  port: int
}
----

==== Request

===== `+Request.method+`

[source,rescript]
----
let method: request => string
----

Get the HTTP method of the request.

*Example*:

[source,rescript]
----
let method = Deno.Request.method(req)
// "GET", "POST", "PUT", etc.
----

===== `+Request.url+`

[source,rescript]
----
let url: request => string
----

Get the full URL of the request.

*Example*:

[source,rescript]
----
let url = Deno.Request.url(req)
// "http://localhost:8000/api/users"
----

===== `+Request.headers+`

[source,rescript]
----
let headers: request => Js.Dict.t<string>
----

Get request headers as a dictionary.

*Example*:

[source,rescript]
----
let headers = Deno.Request.headers(req)
switch Js.Dict.get(headers, "authorization") {
| Some(token) => // Use token
| None => // No auth header
}
----

===== `+Request.text+`

[source,rescript]
----
let text: request => promise<string>
----

Read request body as text.

*Example*:

[source,rescript]
----
let body = await Deno.Request.text(req)
----

===== `+Request.json+`

[source,rescript]
----
let json: request => promise<Js.Json.t>
----

Parse request body as JSON.

*Example*:

[source,rescript]
----
try {
  let json = await Deno.Request.json(req)
  // Process JSON
} catch {
| _ => // Invalid JSON
}
----

==== Response

===== `+Response.text+`

[source,rescript]
----
let text: (
  string,
  ~status: int=?,
  ~headers: Js.Dict.t<string>=?,
  unit
) => response
----

Create a text response.

*Parameters*: - `+data+`: Response body - `+status+`: HTTP status code
(default: 200) - `+headers+`: Additional headers

*Example*:

[source,rescript]
----
Deno.Response.text("Hello, World!", ())
Deno.Response.text("Not Found", ~status=404, ())

let customHeaders = Js.Dict.empty()
Js.Dict.set(customHeaders, "X-Custom", "value")
Deno.Response.text("Data", ~headers=customHeaders, ())
----

===== `+Response.json+`

[source,rescript]
----
let json: (
  Js.Json.t,
  ~status: int=?,
  ~headers: Js.Dict.t<string>=?,
  unit
) => response
----

Create a JSON response.

*Example*:

[source,rescript]
----
let data = Js.Json.object_(Js.Dict.fromArray([
  ("message", Js.Json.string("Success")),
  ("count", Js.Json.number(42.0))
]))

Deno.Response.json(data, ())
Deno.Response.json(data, ~status=201, ())
----

===== `+Response.html+`

[source,rescript]
----
let html: (
  string,
  ~status: int=?,
  ~headers: Js.Dict.t<string>=?,
  unit
) => response
----

Create an HTML response.

*Example*:

[source,rescript]
----
Deno.Response.html("<h1>Hello</h1>", ())
----

=== Server Module

HTTP server creation and management.

==== Types

===== `+config+`

==== Functions

===== `+Server.make+`

[source,rescript]
----
let make: (
  ~port: int=?,
  ~hostname: string=?,
  ~onListen: (Deno.addr => unit)=?,
  Deno.request => promise<Deno.response>
) => unit
----

Create an HTTP server with a request handler.

*Parameters*: - `+port+`: Port number (default: 8000) - `+hostname+`:
Host address (default: "`127.0.0.1`") - `+onListen+`: Callback when
server starts - Handler function: `+request => promise<response>+`

*Example*:

[source,rescript]
----
Server.make(
  ~port=3000,
  ~onListen=addr => Deno.log(`Server on port ${Int.toString(addr.port)}`),
  async req => Deno.Response.text("Hello", ())
)
----

===== `+Server.simple+`

[source,rescript]
----
let simple: (
  ~port: int=?,
  ~hostname: string=?,
  Deno.request => promise<Deno.response>
) => unit
----

Create a simple server (convenience function).

*Example*:

[source,rescript]
----
Server.simple(
  ~port=8000,
  async _req => Deno.Response.text("Hello", ())
)
----

===== `+Server.withRouter+`

[source,rescript]
----
let withRouter: (
  ~port: int=?,
  ~hostname: string=?,
  ~onListen: (Deno.addr => unit)=?,
  Router.t
) => unit
----

Create a server with a router.

*Example*:

[source,rescript]
----
let router = Router.make()
  ->Router.get("/", handler)

Server.withRouter(~port=8000, router)
----

=== Router Module

Type-safe routing and middleware composition.

==== Types

===== `+method+`

[source,rescript]
----
type method = GET | POST | PUT | DELETE | PATCH | OPTIONS | HEAD
----

===== `+route+`

[source,rescript]
----
type route = {
  method: method,
  path: string,
  handler: Deno.request => promise<Deno.response>
}
----

===== `+middleware+`

[source,rescript]
----
type middleware = (
  Deno.request,
  unit => promise<Deno.response>

===== `+t+`

[source,rescript]
----
type t = {
  routes: array<route>,
  middlewares: array<middleware>,

=== Middleware Module

Common middleware implementations.

==== CORS

===== `+Middleware.cors+`

[source,rescript]
----
let cors: (
  ~origin: string=?,
  ~methods: array<string>=?,
  ~headers: array<string>=?,
  ~credentials: bool=?,
  unit
) => Router.middleware
----

CORS middleware.

*Parameters*: - `+origin+`: Allowed origin (default: “*“) - `+methods+`:
Allowed methods (default: ["`GET`", "`POST`", "`PUT`", "`DELETE`",
"`PATCH`"]) - `+headers+`: Allowed headers (default: ["`Content-Type`",
"`Authorization`"]) - `+credentials+`: Allow credentials (default:
false)

*Example*:

[source,rescript]
----
let router = Router.make()
  ->Router.use(Middleware.cors(
    ~origin="https://example.com",
    ~methods=["GET", "POST"],
    ~credentials=true,
    ()
  ))
----

==== Logger

===== `+Middleware.logger+`

[source,rescript]
----
let logger: unit => Router.middleware
----

Request logging middleware.

*Example*:

[source,rescript]
----
let router = Router.make()
  ->Router.use(Middleware.logger())
----

*Output*:

....
GET /api/users - 45.23ms
POST /api/users - 123.45ms
....

==== Rate Limiting

===== `+Middleware.rateLimit+`

[source,rescript]
----
let rateLimit: (
  ~maxRequests: int=?,
  ~windowMs: float=?,
  unit
) => Router.middleware
----

Rate limiting middleware.

*Parameters*: - `+maxRequests+`: Max requests per window (default: 100)
- `+windowMs+`: Time window in milliseconds (default: 60000)

*Example*:

[source,rescript]
----
let router = Router.make()
  ->Router.use(Middleware.rateLimit(
    ~maxRequests=50,
    ~windowMs=30000.0,  // 30 seconds
    ()
  ))
----

==== Authentication

===== `+Middleware.auth+`

[source,rescript]
----
let auth: (~secret: string) => Router.middleware
----

JWT authentication middleware helper.

*Example*:

[source,rescript]
----
let router = Router.make()
  ->Router.use(Middleware.auth(~secret="your-secret-key"))
  ->Router.get("/protected", protectedHandler)
----

==== Error Handling

===== `+Middleware.errorHandler+`

[source,rescript]
----
let errorHandler: unit => Router.middleware
----

Global error handler middleware.

*Example*:

[source,rescript]
----

----

==== Compression

===== `+Middleware.compress+`
